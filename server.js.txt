const shopeeImg = (imgId) => {
  if (!imgId) return null;
  return `https://down-ph.img.susercontent.com/file/${imgId}`;
};

const relevantFallbackImg = (keyword, idx) => {
  const clean = encodeURIComponent(keyword.toLowerCase().replace(/[^a-z0-9 ]/g,' ').trim().split(' ').slice(0,3).join(','));
  return `https://loremflickr.com/400/400/${clean}/all?lock=${idx}${Date.now()%100}`;
};

async function fetchShopeeReal(q, limit = 24) {
  const endpoints = [
    `https://shopee.ph/api/v4/search/search_items?by=relevancy&keyword=${encodeURIComponent(q)}&limit=${limit}&newest=0&order=desc&page_type=search&scenario=PAGE_GLOBAL_SEARCH&version=2`,
    `https://shopee.com.ph/api/v4/search/search_items?by=relevancy&keyword=${encodeURIComponent(q)}&limit=${limit}&newest=0&order=desc&page_type=search&scenario=PAGE_GLOBAL_SEARCH&version=2`
  ];

  for (const url of endpoints) {
    try {
      console.log(`Trying: ${url}`);
      const controller = new AbortController();
      const timeout = setTimeout(() => controller.abort(), 8000);
      const res = await fetch(url, {
        signal: controller.signal,
        headers: {
          'User-Agent': 'Mozilla/5.0 (iPhone; CPU iPhone OS 16_6 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/16_6 Mobile/15E148 Safari/604.1',
          'Accept': 'application/json',
          'Accept-Language': 'en-US,en;q=0.9',
          'Referer': `https://shopee.ph/search?keyword=${encodeURIComponent(q)}`,
          'X-Requested-With': 'XMLHttpRequest',
          'X-Shopee-Language': 'en'
        }
      });
      clearTimeout(timeout);
      console.log(`Status: ${res.status}`);
      if (!res.ok) continue;
      const data = await res.json();
      const items = data?.items || [];
      console.log(`Items: ${items.length}`);
      if (!items.length) continue;

      return items.slice(0, limit).map((wrapper, idx) => {
        const item = wrapper.item_basic || wrapper.item || wrapper;
        const shopid = item.shopid || 0;
        const itemid = item.itemid || 0;
        const name = item.name || q;
        const priceRaw = item.price || item.price_min || 19900000;
        const price = Math.round(priceRaw / 100000);
        const sold = item.sold || item.historical_sold || 1200;
        const rating = item.item_rating?.rating_star || 4.6;
        const reviews = item.item_rating?.rating_count?.[0] || 320;
        const imgId = item.image || '';
        const realImage = imgId ? shopeeImg(imgId) : relevantFallbackImg(name, idx);
        let suriScore = 50 + Math.min(rating*10,45) + Math.min(Math.log10(sold+1)*8,15);
        suriScore = Math.min(Math.round(suriScore), 98);
        return {
          id: `shopee-${shopid}-${itemid}`,
          platform: 'shopee',
          title: name,
          price,
          originalPrice: Math.round(price*1.3),
          sold,
          rating: Number(Number(rating).toFixed(1)),
          reviews,
          image: realImage,
          suriScore,
          url: shopid && itemid ? `https://shopee.ph/-i.${shopid}.${itemid}` : `https://shopee.ph/search?keyword=${encodeURIComponent(name)}`,
          shopid, itemid, isReal: !!imgId
        };
      });
    } catch (err) {
      console.log(`Error: ${err.message}`);
      continue;
    }
  }
  return [];
}

app.get('/api/search', async (req, res) => {
  const q = (req.query.q || '').trim();
  if (!q) return res.status(400).json({ error: 'q required' });
  const limit = parseInt(req.query.limit) || 24;
  console.log(`\n=== SEARCH: "${q}" ===`);
  try {
    let shopeeProducts = await fetchShopeeReal(q, limit);
    const isReal = shopeeProducts.length > 0 && shopeeProducts.some(p => p.isReal);
    console.log(`Real: ${shopeeProducts.length}, isReal: ${isReal}`);

    if (shopeeProducts.length === 0) {
      console.log('Fallback');
      const titles = [`${q} Original`,`${q} Official Store`,`${q} Best Seller`,`${q} Pro Max`,`${q} 20000mAh`,`${q} Wireless`,`${q} Premium`];
      shopeeProducts = Array.from({ length: 12 }, (_, i) => ({
        id: `shopee-fallback-${i}`,
        platform: 'shopee',
        title: titles[i % titles.length],
        price: 299 + i*80,
        originalPrice: 450 + i*80,
        sold: 800 + i*300,
        rating: 4.6,
        reviews: 320,
        image: relevantFallbackImg(q, i),
        suriScore: 80,
        url: `https://shopee.ph/search?keyword=${encodeURIComponent(q)}`,
        isReal: false
      }));
    }

    const lazadaProducts = shopeeProducts.slice(0,12).map((p,i)=>({...p, id:p.id.replace('shopee','lazada'), platform:'lazada', url:`https://www.lazada.com.ph/catalog/?q=${encodeURIComponent(p.title||q)}`, image: p.isReal? p.image : relevantFallbackImg(p.title||q, i+100)}));
    const tiktokProducts = shopeeProducts.slice(0,12).map((p,i)=>({...p, id:p.id.replace('shopee','tiktok'), platform:'tiktok', url:`https://www.tiktok.com/search?q=${encodeURIComponent(p.title||q)}`, image: p.isReal? p.image : relevantFallbackImg(p.title||q, i+200)}));

    return res.json({ query:q, shopee:shopeeProducts, lazada:lazadaProducts, tiktok:tiktokProducts, real:isReal, count:shopeeProducts.length, source: isReal? 'shopee-real-cdn' : 'fallback' });
  } catch (e) {
    console.error(e);
    return res.status(500).json({ error:e.message, shopee:[], lazada:[], tiktok:[], real:false });
  }
});
